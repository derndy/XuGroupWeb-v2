{{- $record := .record -}} {{- $image := resources.Get $record.resource -}} {{- if not $image -}}{{ errorf "Missing published figure: %s" $record.resource }}{{- end -}} {{- if ne (sha256 $image.Content) $record.sha256 -}}{{ errorf "Published figure hash mismatch: %s" $record.resource }}{{- end -}} {{- range slice "caption" "alt" "credit" "source_url" "license" "license_url" -}} {{- if not (index $record .) -}}{{ errorf "Published figure %s lacks %s" $record.resource . }}{{- end -}} {{- end -}} {{- $id := printf "%s-%s" .placement $record.id -}} {{- $sources := slice -}} {{- range slice 480 800 1200 -}} {{- if lt . $image.Width -}} {{- $variant := $image.Resize (printf "%dx webp q90" .) -}} {{- $sources = $sources | append (printf "%s %dw" $variant.RelPermalink $variant.Width) -}} {{- end -}} {{- end -}} {{- $full := $image.Resize (printf "%dx webp q90" $image.Width) -}} {{- $sources = $sources | append (printf "%s %dw" $full.RelPermalink $full.Width) -}}
